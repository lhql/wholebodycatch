cmake_minimum_required(VERSION 3.10)
project(astra_wall_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(gazebo REQUIRED)
add_library(astra_arm_control SHARED src/arm_control.cpp)
target_compile_features(astra_arm_control PRIVATE cxx_std_17)
target_compile_options(astra_arm_control PRIVATE -Wall -Wextra)
target_include_directories(astra_arm_control PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(astra_arm_control PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(astra_arm_control ${GAZEBO_LIBRARIES})
ament_target_dependencies(astra_arm_control rclcpp sensor_msgs std_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS astra_arm_control LIBRARY DESTINATION lib)
install(DIRECTORY launch config worlds models DESTINATION share/${PROJECT_NAME})
install(DIRECTORY docs tests DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS scripts/manual_sliders.py scripts/flight_dashboard.py scripts/manual_teleop.py scripts/air_goal.py scripts/run_sim.sh scripts/seed_parameters.py scripts/wall_control.py scripts/observe.py scripts/configure_estimator.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md PROGRESS.md DESTINATION share/${PROJECT_NAME})
ament_package()
