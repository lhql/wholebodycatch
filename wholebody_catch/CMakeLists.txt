cmake_minimum_required(VERSION 3.10)
project(wholebody_catch)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/catch_vision.py scripts/catch_mission.py scripts/evaluate_catch.py scripts/object_vision.py scripts/make_catch_scene.py
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds DESTINATION share/${PROJECT_NAME})
install(FILES scripts/segment_worker.py DESTINATION share/${PROJECT_NAME}/scripts)
set(CATCH_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../dependencies")
if(EXISTS "${CATCH_DEPS}/sam2/sam2")
  install(DIRECTORY "${CATCH_DEPS}/sam2/sam2" DESTINATION share/${PROJECT_NAME}/vendor/sam2 PATTERN "__pycache__" EXCLUDE)
  install(FILES "${CATCH_DEPS}/sam2/LICENSE" DESTINATION share/${PROJECT_NAME}/vendor/sam2)
  install(DIRECTORY "${CATCH_DEPS}/catch_python/" DESTINATION share/${PROJECT_NAME}/vendor/python311 PATTERN "__pycache__" EXCLUDE)
  install(FILES "${CATCH_DEPS}/catch_models/sam2.1_hiera_tiny.pt" DESTINATION share/${PROJECT_NAME}/weights)
endif()
install(FILES README.md PROGRESS.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(catch_core tests/test_core.py)
  ament_add_pytest_test(catch_generic tests/test_generic.py)
endif()
ament_package()
